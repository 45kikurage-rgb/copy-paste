<script>
const input = document.getElementById('input');
const output = document.getElementById('output');
const blankLine = document.getElementById('blankLine');
const copyBtn = document.getElementById('copyBtn');
const clearBtn = document.getElementById('clearBtn');
const status = document.getElementById('status');
const count = document.getElementById('count');

function normalizeMarkdownUrls(text){
  return text.replace(/\[([^\]]*https?:\/\/[^\]]+)\]\((https?:\/\/[^)]+)\)/g, '$2');
}

function extractPairs(text){
  text = normalizeMarkdownUrls(text)
    .replace(/\r\n?/g, '\n')
    .replace(/\u200b/g, '');

  const lines = text.split('\n').map(s => s.trim()).filter(Boolean);
  const pairs = [];
  let pendingNum = null;

  for (const line of lines){
    const numMatch = line.match(/^(\d+)\s*$/);
    const urlMatch = line.match(/(https?:\/\/[^\s<>"']+)/i);

    if(numMatch){
      pendingNum = numMatch[1];
      continue;
    }

    if(urlMatch){
      pairs.push({num: pendingNum, url: urlMatch[1]});
      pendingNum = null;
      continue;
    }

    // 同じ行に「数字 URL」がある場合
    const sameLine = line.match(/^(\d+)\s+(https?:\/\/[^\s<>"']+)/i);
    if(sameLine){
      pairs.push({num:sameLine[1], url:sameLine[2]});
      pendingNum = null;
    }
  }

  // より自由な形式の保険：URLだけは拾う
  if(pairs.length === 0){
    const urls = text.match(/https?:\/\/[^\s<>"')\]]+/gi) || [];
    urls.forEach((url, i) => pairs.push({num:String(i+1), url}));
  }

  return pairs;
}

function format(){
  const pairs = extractPairs(input.value);
  const sep = blankLine.checked ? '\n\n' : '\n';
  output.value = pairs.map((p, i) => {
    const n = p.num || String(i+1);
    return `${n}${sep}${p.url}`;
  }).join(sep);
  count.textContent = `${pairs.length}件`;
  status.textContent = '';
}

input.addEventListener('input', format);
blankLine.addEventListener('change', format);

clearBtn.addEventListener('click', () => {
  input.value = '';
  output.value = '';
  count.textContent = '0件';
  status.textContent = '';
  input.focus();
});

copyBtn.addEventListener('click', async () => {
  if(!output.value){
    status.textContent = 'コピーする内容がありません。';
    return;
  }
  try{
    await navigator.clipboard.writeText(output.value);
    status.textContent = 'コピーしました！';
  }catch(e){
    output.removeAttribute('readonly');
    output.select();
    document.execCommand('copy');
    output.setAttribute('readonly','');
    status.textContent = 'コピーしました！';
  }
});

format();
</script>
